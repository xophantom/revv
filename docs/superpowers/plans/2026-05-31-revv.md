# Revv Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Revv — a multi-user SaaS for tracking vehicles, maintenance, upgrades, technical spec sheets, smart alerts, and a financial dashboard.

**Architecture:** Next.js App Router with Server Actions for mutations (typed `{ ok, data | error }` results, Zod validation, ownership checks on every call). Prisma + PostgreSQL for persistence. NextAuth v5 for auth with custom login UI. UploadThing for image/PDF uploads (multiple per record). Pure domain functions (alert engine, money, financial aggregation) are unit-tested with Vitest; UI tasks gate on `tsc --noEmit` + `next build`.

**Tech Stack:** Next.js 15 (App Router) · TypeScript · Prisma 6 · PostgreSQL · NextAuth v5 (Auth.js) · UploadThing v7 · Tailwind CSS v4 · shadcn/ui · Framer Motion · Zustand · Recharts · Zod · Vitest.

**Design system (Track Day):** `bg #0a0a0b · surface #141416 · surface2 #1b1c1f · border rgba(255,255,255,.08) · accent #ff6a00 · text #fff · muted rgba(255,255,255,.55)`. Display font **Space Grotesk**, technical data **JetBrains Mono**. Radius 10–14px, subtle glass, Framer Motion entrance/transition.

**Money rule:** stored everywhere as integer cents (`costCents`). Never float. Display BRL.

**Spec reference:** `docs/superpowers/specs/2026-05-31-revv-design.md`

export function realToCents(input: string): number {
  const normalized = input.trim().replace(/\./g, "").replace(",", ".");
  const value = Number(normalized);
  if (Number.isNaN(value)) return 0;
  return Math.round(value * 100);
}
export function centsToBRL(cents: number): string {
  return new Intl.NumberFormat("pt-BR", { style: "currency", currency: "BRL" }).format(cents / 100);
}
```

"use server";
import { z } from "zod";
import { revalidatePath } from "next/cache";
import { db } from "@/lib/db";
import { requireUser } from "@/lib/session";
import { ok, fail, type ActionResult } from "@/lib/result";

const vehicleInput = z.object({
  name: z.string().min(1, "Nome obrigatório"),
  plate: z.string().optional(),
  make: z.string().optional(),
  model: z.string().optional(),
  year: z.coerce.number().int().min(1900).max(2100).optional(),
  engine: z.string().optional(),
  color: z.string().optional(),
  currentKm: z.coerce.number().int().min(0).default(0),
  photoKey: z.string().optional(),
  photoUrl: z.string().url().optional(),
});

async function ownVehicleOr404(id: string, userId: string) {
  const v = await db.vehicle.findUnique({ where: { id } });
  if (!v || v.userId !== userId) throw new Error("not_found");
  return v;
}

export async function createVehicle(raw: unknown): Promise<ActionResult<{ id: string }>> {
  const user = await requireUser();
  const parsed = vehicleInput.safeParse(raw);
  if (!parsed.success) return fail(parsed.error.issues[0].message);
  const v = await db.vehicle.create({ data: { ...parsed.data, userId: user.id } });
  revalidatePath("/garage");
  return ok({ id: v.id });
}

export async function updateVehicle(id: string, raw: unknown): Promise<ActionResult> {
  const user = await requireUser();
  await ownVehicleOr404(id, user.id);
  const parsed = vehicleInput.partial().safeParse(raw);
  if (!parsed.success) return fail(parsed.error.issues[0].message);
  await db.vehicle.update({ where: { id }, data: parsed.data });
  revalidatePath(`/garage/${id}`);
  revalidatePath("/garage");
  return ok(undefined);
}

export async function updateKm(id: string, currentKm: number): Promise<ActionResult> {
  const user = await requireUser();
  await ownVehicleOr404(id, user.id);
  if (!Number.isInteger(currentKm) || currentKm < 0) return fail("km inválido");
  await db.vehicle.update({ where: { id }, data: { currentKm } });
  revalidatePath(`/garage/${id}`);
  return ok(undefined);
}

export async function deleteVehicle(id: string): Promise<ActionResult> {
  const user = await requireUser();
  await ownVehicleOr404(id, user.id);
  await db.vehicle.delete({ where: { id } });
  revalidatePath("/garage");
  return ok(undefined);
}

import type { MaintenanceCategory } from "@prisma/client";
import type { VehicleSpecData } from "./spec-schema";

const KM_SOON = 1500;
const MONTHS_SOON = 1;

export type AlertStatus = "ok" | "soon" | "overdue";
export type Alert = {
  id: string;
  label: string;
  status: AlertStatus;
  remainingKm: number | null;
  remainingMonths: number | null;
  source: "spec" | "custom";
};

type SpecRule = { id: string; label: string; category: MaintenanceCategory; intervalKm?: number; intervalMonths?: number };

export type AlertInput = {
  currentKm: number;
  today: Date;
  spec: VehicleSpecData;
  maintenances: { category: MaintenanceCategory; date: Date; km: number }[];
  customAlerts: { id: string; label: string; dueKm: number | null; dueDate: Date | null; done: boolean }[];
};

function rulesFromSpec(spec: VehicleSpecData): SpecRule[] {
  const r: SpecRule[] = [];
  if (spec.engineOil?.intervalKm || spec.engineOil?.intervalMonths)
    r.push({ id: "engineOil", label: "Troca de óleo", category: "OIL", intervalKm: spec.engineOil.intervalKm, intervalMonths: spec.engineOil.intervalMonths });
  if (spec.tires?.rotationKm) r.push({ id: "tires", label: "Rodízio de pneus", category: "TIRES", intervalKm: spec.tires.rotationKm });
  if (spec.brakeFluid?.intervalMonths) r.push({ id: "brakeFluid", label: "Fluido de freio", category: "FLUIDS", intervalMonths: spec.brakeFluid.intervalMonths });
  if (spec.coolant?.intervalMonths) r.push({ id: "coolant", label: "Arrefecimento", category: "FLUIDS", intervalMonths: spec.coolant.intervalMonths });
  if (spec.sparkPlugs?.intervalKm) r.push({ id: "sparkPlugs", label: "Velas", category: "ELECTRICAL", intervalKm: spec.sparkPlugs.intervalKm });
  if (spec.airFilter?.intervalKm) r.push({ id: "airFilter", label: "Filtro de ar", category: "FILTERS", intervalKm: spec.airFilter.intervalKm });
  if (spec.fuelFilter?.intervalKm) r.push({ id: "fuelFilter", label: "Filtro de combustível", category: "FILTERS", intervalKm: spec.fuelFilter.intervalKm });
  if (spec.cabinFilter?.intervalKm) r.push({ id: "cabinFilter", label: "Filtro de cabine", category: "FILTERS", intervalKm: spec.cabinFilter.intervalKm });
  if (spec.timingBelt?.intervalKm) r.push({ id: "timingBelt", label: "Correia/corrente", category: "REVISION", intervalKm: spec.timingBelt.intervalKm });
  return r;
}

function monthsBetween(a: Date, b: Date): number {
  return (b.getFullYear() - a.getFullYear()) * 12 + (b.getMonth() - a.getMonth());
}

function statusFrom(remainingKm: number | null, remainingMonths: number | null): AlertStatus {
  const overdue = (remainingKm !== null && remainingKm < 0) || (remainingMonths !== null && remainingMonths < 0);
  if (overdue) return "overdue";
  const soon = (remainingKm !== null && remainingKm <= KM_SOON) || (remainingMonths !== null && remainingMonths <= MONTHS_SOON);
  return soon ? "soon" : "ok";
}

const RANK: Record<AlertStatus, number> = { overdue: 0, soon: 1, ok: 2 };

export function computeAlerts(input: AlertInput): Alert[] {
  const alerts: Alert[] = [];
  for (const rule of rulesFromSpec(input.spec)) {
    const last = input.maintenances
      .filter((m) => m.category === rule.category)
      .sort((a, b) => b.date.getTime() - a.date.getTime())[0];
    let remainingKm: number | null = null;
    let remainingMonths: number | null = null;
    if (rule.intervalKm != null) {
      const baseKm = last?.km ?? input.currentKm;
      remainingKm = rule.intervalKm - (input.currentKm - baseKm);
    }
    if (rule.intervalMonths != null) {
      const baseDate = last?.date ?? input.today;
      remainingMonths = rule.intervalMonths - monthsBetween(baseDate, input.today);
    }
    alerts.push({ id: rule.id, label: rule.label, status: statusFrom(remainingKm, remainingMonths), remainingKm, remainingMonths, source: "spec" });
  }
  for (const c of input.customAlerts) {
    if (c.done) continue;
    const remainingKm = c.dueKm != null ? c.dueKm - input.currentKm : null;
    const remainingMonths = c.dueDate != null ? monthsBetween(input.today, c.dueDate) : null;
    alerts.push({ id: c.id, label: c.label, status: statusFrom(remainingKm, remainingMonths), remainingKm, remainingMonths, source: "custom" });
  }
  return alerts.sort((a, b) => RANK[a.status] - RANK[b.status]);
}
```

  const user = await requireUser();

export type FinanceInput = {
  currentKm: number;
  firstKm: number;
  maintenances: { date: Date; costCents: number }[];
  upgrades: { date: Date; costCents: number }[];
};
export type FinanceSummary = {
  maintenanceCents: number;
  upgradeCents: number;
  totalCents: number;
  costPerKmCents: number | null;
  maxExpenseCents: number;
  byMonth: { month: string; cents: number }[];
};

const sum = (xs: { costCents: number }[]) => xs.reduce((a, x) => a + x.costCents, 0);
const ym = (d: Date) => `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`;

export function summarizeVehicle(input: FinanceInput): FinanceSummary {
  const maintenanceCents = sum(input.maintenances);
  const upgradeCents = sum(input.upgrades);
  const totalCents = maintenanceCents + upgradeCents;
  const km = input.currentKm - input.firstKm;
  const all = [...input.maintenances, ...input.upgrades];
  const byMonthMap = new Map<string, number>();
  for (const e of all) byMonthMap.set(ym(e.date), (byMonthMap.get(ym(e.date)) ?? 0) + e.costCents);
  const byMonth = [...byMonthMap.entries()].map(([month, cents]) => ({ month, cents })).sort((a, b) => a.month.localeCompare(b.month));
  return {
    maintenanceCents, upgradeCents, totalCents,
    costPerKmCents: km > 0 ? Math.round(totalCents / km) : null,
    maxExpenseCents: all.reduce((m, e) => Math.max(m, e.costCents), 0),
    byMonth,
  };
}
